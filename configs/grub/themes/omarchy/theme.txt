# Omarchy's boot menu, in Tokyo Night: Plymouth's background (#1a1b26) and
# its logo where Plymouth centres it. After a pick, the screen keeps that
# background until Plymouth draws (configs/grub/grub.cfg).
# builder/boot-menu-theme.sh makes the fonts, the pixmaps and logo.png;
# configs/grub/theme.cfg loads them.
#
# GRUB paints components in the reverse of the order listed here, so what is
# on top comes first.
title-text: ""
desktop-color: "#1a1b26"
terminal-font: "Terminus Bold 16"
terminal-box: "terminal_box_*.png"
terminal-left: "10%"
terminal-top: "10%"
terminal-width: "80%"
terminal-height: "80%"
terminal-border: "0"

# The entries in Tokyo Night's foreground and bright foreground: both read at
# better than 7:1 on the background, and test/integration.d's OCR finds them.
# The installer's console face, Terminus Bold 16 with every pixel doubled
# (builder/grub-font-double.py). Exactly as tall as its two
# entries: 2 x item_height + item_spacing, plus the selection box's 1 px top
# and bottom edges, which gui_list.c takes off before it counts rows. Both
# show on every screen the logo fits.
+ boot_menu {
    left = 50%-150
    top = 65%
    width = 300
    height = 84
    item_font = "Terminus 2x Bold 32"
    selected_item_font = "Terminus 2x Bold 32"
    item_color = "#a9b1d6"
    selected_item_color = "#c0caf5"
    item_height = 40
    item_padding = 0
    item_spacing = 2
    item_icon_space = 16
    icon_width = 0
    icon_height = 0
    selected_item_pixmap_style = "select_*.png"
    scrollbar = true
    scrollbar_width = 4
    scrollbar_frame = "scroll_frame_*.png"
    scrollbar_thumb = "scroll_thumb_*.png"
}

# Centred on screen and sized proportionally so HiDPI panels (e.g. 3840x2400
# laptops where UEFI GOP exposes only native mode) don't draw a tiny 800 px
# logo. 50% x 20% stays within ~6% of the 800:188 aspect ratio on both 16:9
# and 16:10 screens.
+ image {
    left = 25%
    top = 40%
    width = 50%
    height = 20%
    file = "logo.png"
}
