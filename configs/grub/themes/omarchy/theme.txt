# configs/grub/theme.cfg loads them.
#
# GRUB paints components in the reverse of the order listed here, so what is
# on top comes first.

# the selection box's 1 px top and bottom edges, which gui_list.c takes off

    item_padding = 0
    item_spacing = 2
    item_icon_space = 16
    icon_width = 0
    icon_height = 0
    selected_item_pixmap_style = "select_*.png"
    scrollbar = true
    scrollbar_width = 4
    scrollbar_frame = "scroll_frame_*.png"
    scrollbar_thumb = "scroll_thumb_*.png"
}
